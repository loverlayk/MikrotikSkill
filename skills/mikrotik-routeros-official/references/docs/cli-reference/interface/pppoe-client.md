## interface/pppoe-client 
**Type:** Directory

<ArgTable c1="Flag" c2="Name" c3="Description">
<ArgTableRow arg="X" typ="disabled">Whether an item is disabled.</ArgTableRow>
<ArgTableRow arg="I" typ="invalid">Whether the interface is invalid.</ArgTableRow>
<ArgTableRow arg="R" typ="running">Whether the interface is running.</ArgTableRow>
</ArgTable>

<ArgTable c1="Argument" c2="Type" c3="Description">
<ArgTableRow arg="name" typ="string">Name of the PPPoE interface, generated by RouterOS if not specified.</ArgTableRow>
<ArgTableRow arg="max-mtu" typ="num">Maximum Transmission Unit.</ArgTableRow>
<ArgTableRow arg="max-mru" typ="num">Maximum Receive Unit.</ArgTableRow>
<ArgTableRow arg="mrru" typ="num">Maximum packet size that can be received on the link. If a packet is bigger than tunnel MTU, it is split into multiple packets, allowing full size IP or Ethernet packets to be sent over the tunnel.</ArgTableRow>
<ArgTableRow arg="interface" typ="multi { array-id, interface: iface_enum
 }">Interface name on which the client runs.</ArgTableRow>
<ArgTableRow arg="user" typ="string">Username used for authentication.</ArgTableRow>
<ArgTableRow arg="password" typ="string">Password used to authenticate.</ArgTableRow>
<ArgTableRow arg="profile" typ="enum">Specifies which PPP profile configuration is used when establishing the tunnel.</ArgTableRow>
<ArgTableRow arg="keepalive-timeout" typ="enum (disabled) { disabled:0 }">Keepalive timeout in seconds.</ArgTableRow>
<ArgTableRow arg="service-name" typ="string">Specifies the service name set on the access concentrator. Can be left blank to connect to any PPPoE server.</ArgTableRow>
<ArgTableRow arg="ac-name" typ="string">Access Concentrator name. This may be left blank and the client connects to any access concentrator on the broadcast domain.</ArgTableRow>
<ArgTableRow arg="add-default-route" typ="bool">Whether to add a default route automatically.</ArgTableRow>
<ArgTableRow arg="default-route-distance" typ="num">Distance value applied to auto-created default route, if add-default-route is also enabled.</ArgTableRow>
<ArgTableRow arg="dial-on-demand" typ="bool">Connects to AC only when outbound traffic is generated. If selected, a route with gateway address from 10.112.112.0/24 network is added while connection is not established.</ArgTableRow>
<ArgTableRow arg="use-peer-dns" typ="bool">Whether to get DNS settings from the peer.</ArgTableRow>
<ArgTableRow arg="allow" typ="ubit (pap, chap, mschap1, mschap2)">Allowed authentication methods. By default, all methods are allowed.</ArgTableRow>
<ArgTableRow arg="host-uniq" typ="string">Hex formatted host-uniq value.</ArgTableRow>
</ArgTable>
